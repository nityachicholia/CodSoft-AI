
import random

board = [" "] * 9

def show_board():
    print()
    for i in range(0, 9, 3):
        print(" " + " | ".join(board[i:i+3]))
        if i < 6:
            print("---+---+---")
    print()

def check_winner(player):
    wins = [
        (0, 1, 2), (3, 4, 5), (6, 7, 8),
        (0, 3, 6), (1, 4, 7), (2, 5, 8),
        (0, 4, 8), (2, 4, 6)
    ]
    return any(all(board[i] == player for i in line)
               for line in wins)

def computer_move():
    # Try to win if possible
    for i in range(9):
        if board[i] == " ":
            board[i] = "O"
            if check_winner("O"):
                return
            board[i] = " "

    # Block the player's winning move
    for i in range(9):
        if board[i] == " ":
            board[i] = "X"
            if check_winner("X"):
                board[i] = "O"
                return
            board[i] = " "

    # Otherwise choose an available square
    available = [i for i in range(9) if board[i] == " "]
    if available:
        board[random.choice(available)] = "O"

print("Welcome to Tic-Tac-Toe AI!")
print("You are X. Computer is O.")
print("Choose a position from 1 to 9:")

print(" 1 | 2 | 3")
print("---+---+---")
print(" 4 | 5 | 6")
print("---+---+---")
print(" 7 | 8 | 9")

while True:
    show_board()

    try:
        move = int(input("Your move (1-9): ")) - 1
    except ValueError:
        print("Please enter a number from 1 to 9.")
        continue

    if move < 0 or move > 8 or board[move] != " ":
        print("Invalid position! Try again.")
        continue

    board[move] = "X"

    if check_winner("X"):
        show_board()
        print("Congratulations! You win!")
        break

    if " " not in board:
        show_board()
        print("It's a draw!")
        break

    computer_move()

    if check_winner("O"):
        show_board()
        print("Computer wins! Better luck next time.")
        break

    if " " not in board:
        show_board()
        print("It's a draw!")
        break
